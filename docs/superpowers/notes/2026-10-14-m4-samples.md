```swift
public let demoAppGroup = "group.com.andyyhope.featureflag.demo"

public enum DemoEnvironment: String, FlagValue, CaseIterable, FlagValueCases {
    case production
    case staging
    case local
}

public enum CheckoutTier: String, FlagValue, CaseIterable, FlagValueCases {
    case free
    case pro
    case enterprise
}

public enum PaymentKind: String, FlagValue, CaseIterable, FlagValueCases {
    case card
    case wallet
    case transfer
}

@FlagRecord
public struct SpendLimit {
    public var currency: String
    public var maximum: Double
}

@FlagRecord
public struct PaymentMethod {
    @FlagRecordKey public var name: String
    public var kind: PaymentKind
    public var enabled: Bool
    public var minimumSpend: Double
    /// Written with a value, so the payloads that predate it — and every list already
    /// stored on a device — keep working instead of being rejected.
    public var limits: FlagRecords<SpendLimit> = []
}

@FlagContainer(description: "Everything this demo can turn on, and where each value came from")
public struct AppFlags {
    /// Deliberately no `remoteKey`: this flag decides which payload gets fetched, so a
    /// payload must not be able to decide it back.
    @Flag(default: DemoEnvironment.production, description: "Which backend this build talks to")
    public var environment: DemoEnvironment

    @Flag(default: false, description: "Show the redesigned onboarding flow",
          remoteKey: "featureToggles.onboarding.v2")
    public var newOnboarding: Bool

    @Flag(default: 10, description: "How many items to show per page")
    public var pageSize: Int

    @Flag(default: ["AU", "NZ"], description: "Markets the app is live in")
    public var markets: [String]

    @Flag(
        default: [
            PaymentMethod(name: "Visa", kind: .card, enabled: true, minimumSpend: 0),
            PaymentMethod(name: "Apple Pay", kind: .wallet, enabled: true, minimumSpend: 5),
            PaymentMethod(name: "Bank transfer", kind: .transfer, enabled: false, minimumSpend: 50),
        ],
        description: "Payment methods offered at checkout",
        remoteKey: "config.paymentMethods")
    public var paymentMethods: FlagRecords<PaymentMethod>

    @FlagGroup(description: "Checkout")
    public var checkout: CheckoutFlags
}

@FlagContainer
public struct CheckoutFlags {
    @Flag(default: false, description: "Offer Apple Pay at checkout",
          remoteKey: "featureToggles.checkout.applePay")
    public var applePay: Bool

    @Flag(default: CheckoutTier.free, description: "Which pricing tier to present")
    public var tier: CheckoutTier

    @Flag(default: URL(string: "https://api.example.com")!, description: "Backend the app talks to",
          remoteKey: "config.apiEndpoint")
    public var endpoint: URL

    @FlagGroup(description: "Express checkout")
    public var express: ExpressFlags
}

@FlagContainer
public struct ExpressFlags {
    @Flag(default: false, description: "Skip the review step")
    public var oneTap: Bool

    @Flag(default: 3.0, description: "Seconds before the order is placed")
    public var confirmationDelay: Double
}
```

```swift
public enum AppSignal: String, FlagSignal {
    case refetchRemoteConfiguration
    case clearRemoteConfiguration
```

### 2.1 Declaring flags (KSP) - `featureflag-core/src/main/kotlin/com/andyyhope/featureflag/Annotations.kt`

Supported types (`featureflag-ksp/.../TypeMapping.kt:120-152`): Boolean, Int, Long, Double, Float, String, ByteArray, `java.time.Instant`, `java.net.URI` (not `URL`), `List<T>`, `Map<String,T>`, `FlagRecords<R>`, enums. - Generated per container `X`: `XRefs` (open class of `FlagRef`s) and `object XSpec : XRefs, FlagContainerSpec<X>` (see `featureflag-ksp/src/test/resources/expansions/AppFlagsSpec.kt.txt`). Visibility follows the interface (`internal` interface -> `internal` spec).

The `remoteKey` string is arbitrary, so choose between `featureToggles.checkout.googlePay` (consistent rename) or keeping `applePay` in payload paths. ## 3. Bridge permission, as an app sees it

Host:
- Add `implementation(project(":featureflag-bridge"))` (and, transitively, `:featureflag-android` -> `:featureflag-core`). `featureflag-bridge/build.gradle.kts:12` has `api(project(":featureflag-android"))`.
- **The manifest merges automatically**: `featureflag-bridge/src/main/AndroidManifest.xml:5-14` declares `<permission android:name="${applicationId}.permission.FEATURE_FLAG_BRIDGE" android:protectionLevel="signature"/>` and `<provider android:name="com.andyyhope.featureflag.bridge.FlagBridgeProvider" android:authorities="${applicationId}.featureflag" android:exported="true" android:permission="${applicationId}.permission.FEATURE_FLAG_BRIDGE"/>`. No registration. `featureflag-android`'s manifest is an empty `<manifest />`.
- Names derive from the **host's** `applicationId`: authority `<id>.featureflag`, permission `<id>.permission.FEATURE_FLAG_BRIDGE` (`FlagBridgeContract.authority/permission`, `FlagBridgeContract.kt:93,96`). A host `applicationIdSuffix` (such as `.debug`) changes both, and the companion's `hostPackage` must include it (`HostNotFound.message` already warns of this, `FlagBridgeConnection.kt:31-33`).
- The host needs, in code: `SharedPreferencesSource.companion(context)` at the top of the pole's sources; `pole.publishSchema(context)` and signal observers in `Application.onCreate` (class registered with `android:name` in the manifest). The provider refuses callers from its own process, so the host reads through the pole.
- `knownSigner`: documented only in the M2 overview (`docs/superpowers/plans/2026-10-10-m2-overview.md:323`: replace the `<permission>` with `tools:node="replace"`); not tested; M4 should document it. Not needed for the samples.

Companion:
- Only `<uses-permission android:name="<hostId>.permission.FEATURE_FLAG_BRIDGE"/>` (spec "Companion manifest", S6). No `<queries>`. Without it `FlagBridgeClient.connect()` returns `UsesPermissionMissing(permission)` (`FlagBridgeClient.kt:62-64`, message tells the user the line to add). Dependencies: `:featureflag-ui` (api `:featureflag-android`), no bridge module. `featureflag-ui` merges `POST_NOTIFICATIONS`, `POST_PROMOTED_NOTIFICATIONS` and a non-exported `FlagFileProvider` at authority `${applicationId}.featureflag.files` (`featureflag-ui/src/main/AndroidManifest.xml`).
- `FlagBridgeClient.connect()` order (`:61-70`): UsesPermissionMissing -> HostNotFound (`getPackageInfo` fails) -> BridgeNotInstalled (host does not define the permission) -> PermissionDenied (`checkSelfPermission != GRANTED`) -> handshake (`hello`, protocol 1, `openFile(schema)`), which also maps a `SecurityException` to PermissionDenied; else Unresponsive / Incompatible / NotPublished / Connected.
- **`PermissionDenied`** is `FlagBridgeConnection.PermissionDenied(hostPackage: String, hostCertificates: List<String>, companionCertificates: List<String>)` `featureflag-android/.../bridge/FlagBridgeConnection.kt:40-50`; digests are SHA-256 of each signing certificate, uppercase hex, no separators (`FlagBridgeClient.kt:249-279`). Its `message` reads: `'<host>' only lets apps signed with its own key connect. Sign this app with the same key, or list this app's certificate in the host's knownSigner permission (Android 12 and later). Host: <hex>. This app: <hex>.` The `UnavailableScreen(title, message, onRetry)` (`featureflag-ui/.../screens/StatusScreens.kt:32`) shows the message, so the e2e check can read both digests on screen; compare them with `keytool -list -v` / `apksigner verify --print-certs`. The local `~/.android/debug.keystore` (alias `androiddebugkey`, pw `android`) has SHA-256 `7A:10:89:9B:...:92:0A:35`.

How the in-repo tests configure it (`featureflag-bridge/src/androidTest/`): a **single APK** acts as both apps. `AndroidManifest.xml:6` adds `<uses-permission android:name="${applicationId}.permission.FEATURE_FLAG_BRIDGE"/>` (the test app requests the permission it also defines); `:9-13` sets `android:name="...HostTestApplication"` and re-declares the provider with `android:process=":host"` and `tools:node="merge"`. Tests run in the main process as the companion; `HostTestApplication.onCreate` (in `:host` only) sleeps 1 s, writes `BridgeTestHost.schema` via `schema.write(FlagBridgeContract.schemaDirectory(this))` and registers signal observers. So there is **no two-APK test and no signing setup** in the repo today; the cross-signer path (`PermissionDenied` against a real second key) is covered only by Robolectric fakes (`featureflag-android/src/test/.../FlagBridgeClientTest.kt:141-181`). The M4 `untrusted` build is its first real exercise (S6 measured it with throwaway apps: different key -> `acquire…` throws `SecurityException` -> `PermissionDenied` with both digests visible).

## 5. Open questions and risks for the plan

1. **What is `samples/signals`?** No iOS example of that name. Options: (a) a shared Android/JVM library with `AppSignal`/`CacheSignal`, mirroring `project.yml`'s shared `AppSignals.swift` (recommended; the host and the companion both need the same enums); (b) drop it from the table and put the enums in `samples/host`, with the companion duplicating or depending on host (a companion depending on the host module is awkward); (c) it is meant to show a countdown/Live Update (but that lives in `featureflag-ui`). Ask THE USER. The Gradle project names would be `:samples:host`, `:samples:companion` (the spec's e2e uses these) and `:samples:signals`.
2. **applicationIds.** Suggested `com.andyyhope.featureflag.sample.host` and `.sample.companion` (mirrors the iOS `com.andyyhope.featureflag.demoapp` / `.democompanion`). Avoid a host `applicationIdSuffix`: the permission and authority follow the host's applicationId and the companion hard-codes the host package. The companion's `<uses-permission>` name must be written out (or injected by a `manifestPlaceholders` entry from one shared constant) because `FlagBridgeContract.permission()` is a runtime function. Keep one source of truth for the host id (for example a root `gradle.properties` entry such as `featureflag.sample.hostId` feeding the host's `applicationId`, the companion's placeholder and a `BuildConfig` field).
3. **Untrusted companion.** Needs: `buildTypes.create("untrusted") { initWith(getByName("debug")); signingConfig = <second key>; matchingFallbacks += "debug"; applicationIdSuffix = ".untrusted" }`. Without the suffix it cannot coexist with the trusted companion on one device (`INSTALL_FAILED_UPDATE_INCOMPATIBLE`), and the e2e step would have to uninstall first. Its `<uses-permission>` still names the **host's** permission, so keep it in `src/main`. The task `:samples:companion:installUntrusted` follows from AGP's naming for a variant called `untrusted`. Its package name changes `companionCertificates`, not `hostPackage`. Check lint/detekt tasks on the extra variant, and that only the debug variant is unit-tested.
4. **Keystores.** A committed `samples/debug.keystore` (known password) plus `samples/untrusted.keystore`; fine for debug only and never for release. The machine's `~/.android/debug.keystore` is auto-created, so on CI each runner gets its own key: a shared committed key is what makes "same signer" reproducible. Secret scanners may flag the files; add a README warning. Alternatively set `signingConfigs.debug` for both apps to the committed store so default `installDebug` just works. Do not touch `release`.
5. **minSdk / targetSdk.** Libraries are `minSdk 26`, `compileSdk 36`. S6 measured API 36 only; behaviour below API 30 (no package visibility filtering) and with older freezers is unmeasured, and the spec says so. A sample targeting 36 on the `Pixel_7` AVD is the verified path. The ProgressStyle Live Update path needs API 36.x, with a chronometer fallback below.
6. **KSP in an app module.** See 4.3; must verify `android.disallowKotlinSourceSets=false` works with AGP 9.1.1 + KSP 2.2.10-2.0.2 (spec S3 was never ticked), the generated `…Spec` visibility (internal interfaces -> internal specs), and that `allWarningsAsErrors` does not trip on generated code. Library modules do not set `android.disallowKotlinSourceSets` because only `featureflag-core` (JVM) uses KSP today.
7. **Compose convention** is library-typed (section 4.3) - small generalisation needed; the companion app needs Compose, the host sample is also best in Compose (the iOS screen is a SwiftUI `List`), which means both samples use it. A host with no Compose is possible but loses the ported UI. Also `featureflag-ui` debug has `ui-tooling` as `debugImplementation`; an `untrusted` variant will not receive it, which is fine.
8. **Host list in the bridge's androidTest.** `featureflag-ui/build.gradle.kts:11-16` keeps `ui-test-manifest` out of `debugImplementation` so no in-repo debug consumer merges an exported test activity; the sample companion (a debug consumer) is fine because that dependency is `androidTestImplementation` only.
9. **Self-bridging stays refused.** A host cannot embed the companion UI against itself (M2/M3 decision), so the iOS host's "one binary" simplicity does not carry over: two APKs, two installs.
10. **Sample code is not API-frozen but is `internal`-heavy.** The scenario copies in core's tests are `internal` (their module's test source set), so they cannot be shared with the samples; copy them or move them to `samples/host` and make the core scenario test reuse (a reverse dependency the plan should avoid). Duplicating keeps core self-contained.
11. **Payload naming (applePay -> googlePay)** touches: the flag name and so its key (`checkout.google-pay`), `remoteKey` strings in `@Flag` and all three environment payloads + three local payloads, the experiments payload key `checkout.google-pay`, the default record "Apple Pay" and staging payload record, UI strings ("Google Pay"), and the typo payload. Keep the typo (`new-onbaording`) intact.
12. **Signal names.** iOS raw values are lowerCamel; Kotlin enum entries conventionally PascalCase; the wire name is `name`. Either name entries in lowerCamel with a detekt suppression, or accept the PascalCase wire names (nothing external depends on them).
13. **e2e (spec "Verification" 3)** needs the emulator and the sample installs; items depend on APIs checked above: split-screen live update (host pole sees `SharedPreferencesSource` changes through one listener; companion writes arrive via the provider, same process as the host), "Source = Companion" (the `SharedPreferencesSource` name; do not rename it), `adb shell am kill <host>` then a signal (cold start; relies on the startup barrier), QR/import round trip, bad JSON import rejected in full, `installUntrusted` shows `PermissionDenied`. M3 deviation 10 moved this manual check to M4. Remember the M3 note on the countdown notification: `POST_NOTIFICATIONS` is requested at the first delayed send (API 33+), so grant it before screenshot runs.
14. **Remaining M4 items outside this plan:** `docs-samples` + `DocsSyncTest`, README/`docs/*.md`, CI, `apiCheck`, publishing (all in the spec's M4 checklist, `specs/...:264-268`), `knownSigner` documentation, `FlagPole.apply`/naming drift decisions before the API freeze (`notes/2026-10-09-m1-followups.md:34-55`).
15. **Stale iOS README line** about `EnvironmentTab.swift` (section 1) - do not port it; the Android README should say the Environment tab is a `Detail(key = "environment")`.
