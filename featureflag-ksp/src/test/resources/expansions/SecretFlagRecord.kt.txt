package app

import com.andyyhope.featureflag.records.FlagRecordCodec
import com.andyyhope.featureflag.records.FlagRecordField
import com.andyyhope.featureflag.records.FlagRecords
import com.andyyhope.featureflag.values.FlagCodec
import com.andyyhope.featureflag.values.FlagCodecs
import com.andyyhope.featureflag.values.FlagValueBox
import kotlin.Exception
import kotlin.IllegalStateException
import kotlin.Int
import kotlin.String
import kotlin.collections.List
import kotlin.collections.Map
import kotlin.collections.buildMap
import kotlin.collections.firstOrNull
import kotlin.collections.listOf
import kotlin.lazy
import kotlin.let

internal object SecretFlagRecord : FlagRecordCodec<Secret> {
  private val idCodec: FlagCodec<String> = FlagCodecs.string

  private val levelCodec: FlagCodec<Int> = FlagCodecs.int

  override val keyField: String? = "id"

  override val shape: List<FlagRecordField> by lazy {
    val defaults = prototype()
    listOf(
      FlagRecordField.of("id", idCodec, isKey = true),
      FlagRecordField.of("level", levelCodec, default = defaults.level),
    )
  }

  override fun boxes(record: Secret): Map<String, FlagValueBox> = buildMap {
    put("id", idCodec.encode(record.id))
    put("level", levelCodec.encode(record.level))
  }

  override fun fromBoxes(boxes: Map<String, FlagValueBox>): Secret? {
    val record = Secret(
      id = boxes["id"]?.let(idCodec::decode) ?: return null,
      level = boxes["level"]?.let(levelCodec::decode) ?: return null,
    )
    return record
  }

  private fun prototype(): Secret = try {
    Secret(id = "")
  } catch (e: Exception) {
    throw IllegalStateException("Could not read the field defaults of 'Secret': its constructor threw when 'SecretFlagRecord' called it with placeholder values (false, 0, \"\", empty, an enum's first case) for the fields that have no default. Give those fields defaults the constructor accepts.", e)
  }
}

internal operator fun FlagRecords<Secret>.`get`(key: String): Secret? {
  val codec = FlagCodecs.string
  val wanted = codec.encode(key)
  return firstOrNull { codec.encode(it.id) == wanted }
}
